cmake -S . -B build-release -DCMAKE_BUILD_TYPE=Release
cmake --build build-release --target test_fisher_kolmogorov_fornari83

# The three scalings 0.04, 0.03 and 0.025 locate the validity boundary of
# the FEM between rho = 0.05 (bounded) and rho = 0.04 (unbounded).
python3 scripts/study-fisher-diffusion-scaling.py \
  --executable build-release/test_fisher_kolmogorov_fornari83 \
  --output-dir output/fisher_kolmogorov/diffusion_scaling \
  --scalings 1.0 0.5 0.2 0.1 0.05 0.04 0.03 0.025 0.02 0.01 0.005 0.002 0.001

# Record figure: the nodal biomarkers at three scalings plus the spread
# against Da over the whole sweep.
python3 scripts/plot-fisher-diffusion-scaling.py \
  output/fisher_kolmogorov/diffusion_scaling \
  --output benchmarks/23_fisher_kolmogorov_diffusion_scaling/results/diffusion_scaling_spread.png

cp output/fisher_kolmogorov/diffusion_scaling/diffusion_scaling.csv \
   output/fisher_kolmogorov/diffusion_scaling/diffusion_scaling_summary.json \
   benchmarks/23_fisher_kolmogorov_diffusion_scaling/results/

# Transient behind the recoverable-violation statement: entorhinal seed at
# rho = 0.05 (Da = 12.9), whose biomarker extremes are stored in results.

cp output/fisher_kolmogorov/staging/tau/fem_biomarkers.csv benchmarks/23_fisher_kolmogorov_diffusion_scaling/results/fem_transient_rho_0p05.csv

# Single runs behind the FEM validity statement.
./build-release/test_fisher_kolmogorov_fornari83 1 0.4 output/scratch 0.5 40 0.005 cn
./build-release/test_fisher_kolmogorov_fornari83 1 0.4 output/scratch 0.5 40 0.005 be

# Stabilized variant: the mass-lumped FEM (row-sum mass, vertex-rule
# reaction) over the whole sweep with both schemes, plus the four-element
# case at rho = 0.005 where the consistent mass diverged.
for rho in 1.0 0.5 0.2 0.1 0.05 0.04 0.03 0.025 0.02 0.01 0.005 0.002 0.001; do
  for scheme in be_lumped cn_lumped; do
    ./build-release/test_fisher_kolmogorov_fornari83 1 0.4 \
      output/fisher_kolmogorov/lumped/rho_${rho}_${scheme} 0.5 120 $rho $scheme
  done
done
for scheme in be_lumped cn_lumped; do
  ./build-release/test_fisher_kolmogorov_fornari83 4 0.4 \
    output/fisher_kolmogorov/lumped/rho_0.005_cells4_${scheme} 0.5 120 0.005 $scheme
done
python3 scripts/tabulate-fem-lumped-sweep.py \
  --output benchmarks/23_fisher_kolmogorov_diffusion_scaling/results/fem_lumped_sweep.csv

# Report figure: nodal model, FEM with consistent mass and FEM with lumped
# mass at three scalings, same mesh, scheme and parameters (fully implicit,
# one element per connection, T = 40); the lobes separate in all three, the
# consistent mass leaves [0,1] and breaks down, the lumped mass does not.
for rho in 1.0 0.05 0.005; do
  for scheme in nodal be be_lumped; do
    ./build-release/test_fisher_kolmogorov_fornari83 1 0.4 \
      output/fisher_kolmogorov/stabilization/rho_${rho}_${scheme} 0.5 40 $rho $scheme
  done
done
python3 scripts/plot-fisher-stabilization.py \
  --output benchmarks/23_fisher_kolmogorov_diffusion_scaling/results/diffusion_scaling.png

# The Damkohler number of the finite element model against the nodal one:

python3 scripts/tabulate-connectome-mass-spectrum.py
