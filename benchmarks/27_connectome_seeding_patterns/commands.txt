cmake --build build-release --target test_fisher_kolmogorov_fornari83

# Tau seeding (entorhinal) and amyloid-beta seeding (neocortex proper, 54
# vertices), finite element model with the lumped mass and the fully
# implicit scheme, at the scaling that reproduces the published separation.
./build-release/test_fisher_kolmogorov_fornari83 1 0.4 output/fisher_kolmogorov/staging/tau_lumped 0.5 40 0.005 be_lumped
./build-release/test_fisher_kolmogorov_fornari83 1 0.4 output/fisher_kolmogorov/staging/amyloid_lumped 0.5 40 0.005 be_lumped neocortex

# The tau run at rho = 1, the transport scale of the references, stored to
# show that no staging exists there: at the first stage every vertex lies
# between 0.10 and 0.13 and the four lobes cross together.
./build-release/test_fisher_kolmogorov_fornari83 1 0.4 output/fisher_kolmogorov/staging/tau_lumped_rho1 0.5 40 1 be_lumped
cp output/fisher_kolmogorov/staging/tau_lumped_rho1/fem_profiles.csv benchmarks/27_connectome_seeding_patterns/results/tau_rho1_profiles.csv
cp output/fisher_kolmogorov/staging/tau_lumped_rho1/fem_biomarkers.csv benchmarks/27_connectome_seeding_patterns/results/tau_rho1_biomarkers.csv

# The same two seedings with the regional reaction rates of Corti et al.,
# rescaled to the same vertex mean 0.5, so that only the distribution of the
# reaction changes: the added biology of Section 4.8 applied to the clinical
# seedings. The horizon is 80 years because the slow groups need it.
./build-release/test_fisher_kolmogorov_fornari83 1 0.4 output/fisher_kolmogorov/staging/tau_regional 0.5 80 0.005 be_lumped -1 benchmarks/21_fisher_kolmogorov_corti83/results/reaction_coefficients.csv
./build-release/test_fisher_kolmogorov_fornari83 1 0.4 output/fisher_kolmogorov/staging/amyloid_regional 0.5 80 0.005 be_lumped neocortex benchmarks/21_fisher_kolmogorov_corti83/results/reaction_coefficients.csv

python3 scripts/plot-connectome-staging.py \
  --tau output/fisher_kolmogorov/staging/tau_regional/fem_profiles.csv \
  --amyloid output/fisher_kolmogorov/staging/amyloid_regional/fem_profiles.csv \
  --reference-dir benchmarks/27_connectome_seeding_patterns/reference \
  --output-dir benchmarks/27_connectome_seeding_patterns/results \
  --prefix seeding_patterns_regional

# The tau row alone: the regional rates restore the clinical lobe order for
# tau, while for amyloid-beta they add an anterior gradient inside the seeded
# mantle that the staging drawings do not show (see the README).
python3 scripts/plot-connectome-staging.py \
  --tau output/fisher_kolmogorov/staging/tau_regional/fem_profiles.csv \
  --amyloid output/fisher_kolmogorov/staging/amyloid_regional/fem_profiles.csv \
  --reference-dir benchmarks/27_connectome_seeding_patterns/reference \
  --output-dir benchmarks/27_connectome_seeding_patterns/results \
  --prefix seeding_patterns_regional_tau --only tau

# Control: the same tau seeding with a synthetic rate field, linear in the
# anterior-posterior coordinate and sharing with the reference only the ratio
# 3.3 between its fastest and its slowest vertex. It answers the objection
# that the coefficients of Corti et al. are estimated for amyloid-beta.
python3 scripts/make-synthetic-rate-field.py \
  --output benchmarks/27_connectome_seeding_patterns/results/synthetic_rate_field.csv
./build-release/test_fisher_kolmogorov_fornari83 1 0.4 output/fisher_kolmogorov/staging/tau_synthetic 0.5 80 0.005 be_lumped -1 benchmarks/27_connectome_seeding_patterns/results/synthetic_rate_field.csv
cp output/fisher_kolmogorov/staging/tau_synthetic/fem_biomarkers.csv \
   benchmarks/27_connectome_seeding_patterns/results/tau_synthetic_biomarkers.csv

# What a regional field must satisfy: the seven values assigned to the seven
# groups in all 5040 ways, with the order of the four lobes recorded.
python3 scripts/study-rate-permutations.py \
  --output benchmarks/27_connectome_seeding_patterns/results/rate_permutations.csv

for case in tau amyloid; do
  cp output/fisher_kolmogorov/staging/${case}_regional/fem_profiles.csv \
     benchmarks/27_connectome_seeding_patterns/results/${case}_regional_profiles.csv
  cp output/fisher_kolmogorov/staging/${case}_regional/fem_biomarkers.csv \
     benchmarks/27_connectome_seeding_patterns/results/${case}_regional_biomarkers.csv
done

# The earlier nodal-model variant of the same two runs, kept for the record.

./build-release/test_fisher_kolmogorov_fornari83 1 0.4 output/fisher_kolmogorov/staging/amyloid_nodal 0.5 40 0.005 nodal neocortex
